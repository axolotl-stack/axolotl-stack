# Buckets and spawn eggs against BDS: `python tools/bds-sweep/script.py
# tools/bds-sweep/scripts/buckets-eggs.txt --out DIR`. Every use is a click
# (`use-on X Y Z FACE`) with the item in hotbar slot 0, survival unless a
# step says otherwise. Ticks are the client's inputs.
#
# Arena as in breaking.txt: a stone floor at y=-61 over dirt, the player at
# (0.5,-60,0.5) facing east, put back before every step. Liquids go in a
# one-block pit in the floor at (2,-61,0), so they cannot flow; the client
# learns the blocks it clicks from the corrections BDS sends for a first,
# empty-handed click.
20 /gamerule sendcommandfeedback false
21 /gamerule domobspawning false
22 /gamerule dodaylightcycle false
23 /gamerule randomtickspeed 0
24 /gamerule dofiretick false
25 /time set 6000
26 /fill -4 -61 -4 4 -61 4 stone
27 /fill -4 -60 -4 4 -55 4 air
28 /gamemode survival @s
29 /clear @s
30 /tp @s 0.5 -60 0.5 -90 0
31 /effect @s clear
32 /setblock 2 -61 0 air
36 use-on 2 -62 0 1

# 1. A water bucket on the pit's floor (dirt, top face); an empty bucket on
#    the water; an empty bucket on the dirt under water put back by command.
40 /replaceitem entity @s slot.weapon.mainhand 0 water_bucket
45 use-on 2 -62 0 1
65 /replaceitem entity @s slot.weapon.mainhand 0 bucket
70 use-on 2 -61 0 1
90 /setblock 2 -61 0 water
95 /replaceitem entity @s slot.weapon.mainhand 0 bucket
100 use-on 2 -62 0 1
120 /setblock 2 -61 0 air
121 /clear @s

# 2. Lava the same way; three empty buckets on the lava (one becomes a lava
#    bucket, the others stay).
130 /replaceitem entity @s slot.weapon.mainhand 0 lava_bucket
135 use-on 2 -62 0 1
155 /replaceitem entity @s slot.weapon.mainhand 0 bucket 3
160 use-on 2 -61 0 1
180 /setblock 2 -61 0 air
181 /clear @s

# 3. Powder snow on the floor's top face at (3,-61,0) (it lands at
#    (3,-60,0)), taken back by clicking it.
185 use-on 3 -61 0 1
195 /replaceitem entity @s slot.weapon.mainhand 0 powder_snow_bucket
200 use-on 3 -61 0 1
220 /replaceitem entity @s slot.weapon.mainhand 0 bucket
225 use-on 3 -60 0 1
245 /setblock 3 -60 0 air
246 /clear @s

# 4. Water into a bottom stone slab in the pit (waterlogs it), then taken
#    back.
250 /setblock 2 -61 0 stone_block_slab
252 use-on 2 -61 0 1
255 /replaceitem entity @s slot.weapon.mainhand 0 water_bucket
260 use-on 2 -61 0 1
280 /replaceitem entity @s slot.weapon.mainhand 0 bucket
285 use-on 2 -61 0 1
305 /setblock 2 -61 0 air
306 /clear @s

# 5. Milk drunk to the end.
310 /effect @s speed 30 0
315 /replaceitem entity @s slot.weapon.mainhand 0 milk_bucket
320 use
360 /clear @s
362 /tp @s 0.5 -60 0.5 -90 0
