# Throwables against BDS: `python tools/bds-sweep/script.py
# tools/bds-sweep/scripts/throwables.txt --out DIR`. Every throw is an item
# use in the air (`use`) with the item given to hotbar slot 0. Ticks are
# the client's inputs.
#
# Arena as in breaking.txt: a stone floor at y=-61, the player at
# (0.5,-60,0.5) facing east. A throw "ahead" flies east, level; a throw
# "down" (pitch 90) lands at the player's feet.

21 /gamerule domobspawning false
22 /gamerule dodaylightcycle false
23 /gamerule randomtickspeed 0
24 /gamerule dofiretick false
25 /time set 6000
26 /fill -4 -61 -4 4 -61 4 stone
27 /fill -4 -60 -4 4 -55 4 air
28 /gamemode survival @s
29 /clear @s
30 /tp @s 0.5 -60 0.5 -90 0
31 /effect @s clear
